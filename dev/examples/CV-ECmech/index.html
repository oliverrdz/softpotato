    import matplotlib.pyplot as plt

from softpotato.geometry import PlanarGeometry
from softpotato.reactions import (
    ButlerVolmer,
    ElectrochemicalStep,
    FirstOrderReaction,
    Mechanism,
    Species,
)
from softpotato.techniques import CyclicVoltammetry, simulate

# Constants
D = 1e-9  # m^2/s
cR_bulk = 1  # mol/m^3 (1 mM)
cO_bulk = 0  # mol/m^3
cP_bulk = 0  # mol/m^3
area = 1e-4  # m^2 (1 cm^2)
E0 = 0  # V
n_rxn = 1
kc = 1  # 1/s
k0 = 1  # m/s
alpha = 0.5

# Define CV
cv = CyclicVoltammetry(E_start=-0.4, E_vertex1=0.4, E_vertex2=-0.4, scan_rate=0.1)

# Species definition
red = Species(
    name="Red",
    diffusivity=D,
    bulk_concentration=cR_bulk,
    standard_potential=E0,
    n_electrons=-n_rxn,
)

ox = Species(
    name="Ox",
    diffusivity=D,
    bulk_concentration=cO_bulk,
    standard_potential=E0,
    n_electrons=n_rxn,
)

prod = Species(name="Prod", diffusivity=D, bulk_concentration=cP_bulk, n_electrons=0)

ec_mech = Mechanism(
    species=[ox, red, prod],
    reactions=[FirstOrderReaction(ox, prod, k=kc)],
    electrochemical_steps=[
        ElectrochemicalStep(
            ox=ox, red=red, kinetics=ButlerVolmer(k0=k0, alpha=alpha, E0=E0, n=n_rxn)
        )
    ],
)

# Set planar geometry
geometry = PlanarGeometry(area)

# Run simulation
# The mechanism is written in the order of the reaction: Ox + e- <-> Red
sim = simulate(technique=cv, mechanism=ec_mech, geometry=geometry)
