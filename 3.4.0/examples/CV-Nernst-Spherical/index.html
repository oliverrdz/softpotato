    import matplotlib.pyplot as plt
import numpy as np

from softpotato.geometry import SphericalGeometry
from softpotato.reactions import Species
from softpotato.techniques import CyclicVoltammetry, simulate

# Constants
D = 1e-9  # m^2/s
cR_bulk = 1  # mol/m^3 (1 mM)
cO_bulk = 0  # mol/m^3
r0 = 1e-5  # m (10 µm)
E0 = 0  # V
n_rxn = 1

# Define CV
cv = CyclicVoltammetry(E_start=-0.4, E_vertex1=0.4, E_vertex2=-0.4, scan_rate=1e-3)

# Species definition
red = Species(
    name="Red",
    diffusivity=D,
    bulk_concentration=cR_bulk,
    standard_potential=E0,
    n_electrons=-n_rxn,
)

ox = Species(
    name="Ox",
    diffusivity=D,
    bulk_concentration=cO_bulk,
    standard_potential=E0,
    n_electrons=n_rxn,
)

# Set planar geometry
geometry = SphericalGeometry(r0=1e-4)

# Run simulation
# The mechanism is written in the order of the reaction: Ox + e- <-> Red
sim = simulate(technique=cv, mechanism=[ox, red], geometry=geometry)
